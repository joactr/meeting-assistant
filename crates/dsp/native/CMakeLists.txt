cmake_minimum_required(VERSION 3.20)
project(meeting_speexdsp C)
if(NOT DEFINED SPEEX_SOURCE)
  message(FATAL_ERROR "SPEEX_SOURCE must name the verified SpeexDSP 1.2.1 extraction")
endif()
file(TO_CMAKE_PATH "${SPEEX_SOURCE}" SPEEX_SOURCE)
set(SOURCES preprocess.c jitter.c mdf.c fftwrap.c filterbank.c resample.c buffer.c scal.c smallft.c)
list(TRANSFORM SOURCES PREPEND "${SPEEX_SOURCE}/libspeexdsp/")
add_library(speexdsp SHARED ${SOURCES})
target_include_directories(speexdsp PRIVATE "${SPEEX_SOURCE}/include" "${SPEEX_SOURCE}/libspeexdsp" "${CMAKE_CURRENT_BINARY_DIR}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.h" "#define FLOATING_POINT 1\n#define USE_SMALLFT 1\n#define EXPORT\n#define SPEEXDSP_EXPORT\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/speexdsp_config_types.h" "#include <stdint.h>\ntypedef int16_t spx_int16_t;\ntypedef uint16_t spx_uint16_t;\ntypedef int32_t spx_int32_t;\ntypedef uint32_t spx_uint32_t;\n")
target_compile_definitions(speexdsp PRIVATE HAVE_CONFIG_H FLOATING_POINT USE_SMALLFT)
set_target_properties(speexdsp PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON SOVERSION 1)
if(MSVC)
 target_compile_definitions(speexdsp PRIVATE _CRT_SECURE_NO_WARNINGS inline=__inline)
else()
 target_link_libraries(speexdsp PRIVATE m)
endif()
install(TARGETS speexdsp RUNTIME DESTINATION . LIBRARY DESTINATION .)
